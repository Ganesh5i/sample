3. Histogram Equalization of a Color Image

clc;
clear;
close all;

% Read the color image
img = imread('rose.webp');

% Display original image
figure;

subplot(2,2,1);
imshow(img);
title('Original Color Image');

% Convert RGB image to HSV
hsvImg = rgb2hsv(img);

% Extract the Value (Brightness) channel
V = hsvImg(:,:,3);

% Perform Histogram Equalization on Value channel
Veq = histeq(V);

% Replace the Value channel with equalized one
hsvImg(:,:,3) = Veq;

% Convert HSV back to RGB
equalizedImg = hsv2rgb(hsvImg);

% Display equalized color image
subplot(2,2,2);
imshow(equalizedImg);
title('Histogram Equalized Color Image');

% Display histogram of original Value channel
subplot(2,2,3);
imhist(V);
title('Original Value Histogram');

% Display histogram of equalized Value channel
subplot(2,2,4);
imhist(Veq);
title('Equalized Value Histogram');

4. Edge Detection Using Sobel Gradient Operator

clc;
clear;
close all;

% Read input image
I = imread('rose.jpg');

% Convert RGB image to grayscale
if size(I,3) == 3
    Igray = rgb2gray(I);
else
    Igray = I;
end

% Convert image to double
Igray = im2double(Igray);

% Sobel gradient masks
Gx = [-1 0 1;
      -2 0 2;
      -1 0 1];

Gy = [-1 -2 -1;
       0  0  0;
       1  2  1];

% Calculate gradients
Ix = imfilter(Igray, Gx, 'replicate');
Iy = imfilter(Igray, Gy, 'replicate');

% Calculate gradient magnitude
G = sqrt(Ix.^2 + Iy.^2);

% Normalize gradient
G = mat2gray(G);

% Sharpen image using gradient
alpha = 1.0;
Sharpened = Igray + alpha * G;

% Limit intensity values to [0,1]
Sharpened = min(max(Sharpened,0),1);

% Display results
figure;

subplot(2,2,1);
imshow(Igray);
title('Original Grayscale Image');

subplot(2,2,2);
imshow(Ix,[]);
title('Gradient in X Direction');

subplot(2,2,3);
imshow(G,[]);
title('Gradient Magnitude');

subplot(2,2,4);
imshow(Sharpened);
title('Sharpened Image');

6. Image Segmentation

clc;
clear;
close all;

% Read input image
img = imread('fig.jpg');

% Convert RGB image to grayscale
gray = rgb2gray(img);

% Set threshold value
threshold = 20;

% Create binary mask
mask = gray > threshold;

% Remove small regions
mask = bwareaopen(mask, 500);

% Detect boundary
boundary = bwperim(mask);

% Copy original image
marked_img = img;

% Extract RGB channels
R = marked_img(:,:,1);
G = marked_img(:,:,2);
B = marked_img(:,:,3);

% Mark boundary in green
R(boundary) = 0;
G(boundary) = 255;
B(boundary) = 0;

% Update RGB channels
marked_img(:,:,1) = R;
marked_img(:,:,2) = G;
marked_img(:,:,3) = B;

% Display original image
subplot(1,3,1);
imshow(img);
title('Original Image');

% Display segmented region
subplot(1,3,2);
imshow(mask);
title('Segmented Region');

% Display segmented image with boundary
subplot(1,3,3);
imshow(marked_img);
title('Segmented Image with Boundary');
